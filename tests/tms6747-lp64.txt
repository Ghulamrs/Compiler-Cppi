# Cases whose expected output was written by a 64-bit-long, 8-byte-pointer
# host, or that assert one: the C6000 has neither, so tests/tms6747.sh skips
# these by name. Each is listed with what it measures.
auto                  sizeof a pointer in the printed sizes
base-tail-padding     static_assert on a layout with an 8-byte vptr
bitfield-zero-width   a bit-field in a 64-bit unit (unsigned long long), refused by the backend
include-c-wrappers    sizeof a pointer
nullptr               sizeof nullptr_t
own-vptr-plain-base   sizeof a class with an 8-byte vptr
reference-member      sizeof a class holding a reference
static-assert         asserts that the target is 64-bit
static-cast           a 40-bit value in a long
vtable                static_assert on a layout with an 8-byte vptr
